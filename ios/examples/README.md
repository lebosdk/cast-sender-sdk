# Examples

- `ObjC/LBCastCoordinator.h/.m`: media push coordinator.
- `ObjC/LBQRScanViewController.h/.m`: QR scan page (AVFoundation); pass `onScan`'s string to `[cast connectWithQRCode:]`. Needs `integrate_sdk.rb --camera`.
- `Mirror/LBMirrorController.h/.m`: App-side mirroring (needs `LeboMirrorConfig.h` generated by `integrate_mirror.rb`). Call `setup` after auth, then `startWithConnection:[LBCastCoordinator shared].connection`.
- `Mirror/SampleHandler.h/.m`: broadcast extension template; `integrate_mirror.rb` copies it with the App Group filled in.

`ObjC/LBCastCoordinator.h/.m` is a complete coordinator: auth, search, connect, push, error routing with the install guide. Copy both files into the App target and adapt the callbacks to the app's UI.

```objc
#import "LBCastCoordinator.h"
#import "SkillSdkConfig.h"

LBCastCoordinator *cast = [LBCastCoordinator shared];
cast.onChange = ^{ /* reload the device list from cast.services; show cast.connectedService */ };
cast.onInstallGuide = ^(NSDictionary *info) { /* open info[LBSkillInstallGuideURLKey], fall back to info[LBSkillInstallGuideStepsKey] */ };
cast.onError = ^(NSError *error) { /* show error.domain + error.code; -1701 means check Wi-Fi / local network */ };
cast.onPlayStatus = ^(LBLelinkPlayStatus status) {
    if (status == LBLelinkPlayStatusPlaying) {
        LBCastCapability *cap = [cast capability];
        // only the controls the user picked; each one follows its flag, stop is always shown
        pauseButton.hidden = !cap.canPause;          // [cast pause] / [cast resumePlay]
        progressSlider.hidden = !cap.canSeek;        // [cast seekTo:seconds]
        volumeSlider.hidden = !cap.canVolume;        // [cast setVolume:0...100]
        rateButton.hidden = !cap.canRate;            // options from cap.rates → [cast setPlaySpeed:rate]
        episodeButtons.hidden = !cap.canSwitchItem;  // [cast playNext] / [cast playPrevious] / [cast playMediaId:]
        clarityHint.hidden = !cap.canDefinition;     // switched on the TV; fill itemClarityArray when pushing the list
    }
};
cast.onProgress = ^(NSInteger current, NSInteger duration) { /* update progressSlider */ };
cast.onCurrentMediaId = ^(NSString *mediaId) { /* highlight the current episode */ };
[cast enableLogConsole:YES file:YES]; // the log items the user picked, before auth
[cast startWithAppId:SkillSdkAppId secretKey:SkillSdkAppSecret];
// on user selection: [cast connect:cast.services[i]];
// cross-network: [cast connectWithCastCode:@"<code on the TV>"]; or push LBQRScanViewController and call [cast connectWithQRCode:value] in onScan
// after onChange shows connectedService: [cast playOnlineVideo:@"https://..."];
```

## Swift

The same flow in Swift. The delegate protocols require `NSObject`.

```swift
import LBLelinkKit

final class CastCoordinator: NSObject, LBLelinkBrowserDelegate, LBLelinkConnectionDelegate, LBLelinkPlayerDelegate {
    static let shared = CastCoordinator()
    private(set) var services: [LBLelinkService] = []
    private var browser: LBLelinkBrowser?
    private let connection = LBLelinkConnection()
    private let player = LBLelinkPlayer()

    func start() {
        connection.delegate = self
        player.delegate = self
        DispatchQueue.global().async {
            do {
                try LBLelinkKit.auth(withAppid: SkillSdkConfig.appId, secretKey: SkillSdkConfig.appSecret)
            } catch {
                return // 402: secretKey must be md5Key; 1002: 180 s cooldown after a failure
            }
            DispatchQueue.main.async {
                let browser = LBLelinkBrowser()
                browser.delegate = self
                // Bluetooth discovery is off by default; the SDK declares onError as non-optional and ignores it
                browser.enableSearchLelinkService(true, sourceStyle: .searchBluetooth, onError: NSError(domain: "", code: 0))
                browser.searchForLelinkService(of: [.lelink, .dlnaDMC, .publickNetwork])
                self.browser = browser
            }
        }
    }

    func connect(_ service: LBLelinkService) {
        connection.lelinkService = service
        connection.connect()
    }

    func play(_ url: String) {
        let item = LBLelinkPlayerItem()
        item.mediaType = .videoOnline
        item.mediaURLString = url
        player.play(with: item)
    }

    func lelinkBrowser(_ browser: LBLelinkBrowser, didFind services: [LBLelinkService]?) {
        self.services = services ?? []
    }

    func lelinkBrowser(_ browser: LBLelinkBrowser, onError error: Error?) {
        guard let error = error as NSError? else { return }
        if let url = error.userInfo[LBSkillInstallGuideURLKey] as? String { _ = url /* show install guide */ }
    }

    // Required by the protocols
    func lelinkBrowser(_ browser: LBLelinkBrowser, permissionsPolicyDenied systemPermissionsType: LBLelinkSystemPermissionsType) {}
    func lelinkConnectionDidStartRetryConnect(_ connection: LBLelinkConnection) {}

    func lelinkConnection(_ connection: LBLelinkConnection, didConnectTo service: LBLelinkService) {
        player.lelinkConnection = connection // must be reassigned after every successful connect
    }
    func lelinkConnection(_ connection: LBLelinkConnection, onError error: Error?) {}
    func lelinkPlayer(_ player: LBLelinkPlayer, playStatus: LBLelinkPlayStatus) {}
    func lelinkPlayer(_ player: LBLelinkPlayer, onError error: Error?) {}
}
```

Swift names are generated from the Objective-C headers. If one does not compile, check the generated interface of `LBLelinkKit.h` in Xcode and use its spelling.
